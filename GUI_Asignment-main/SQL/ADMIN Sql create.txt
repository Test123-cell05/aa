CREATE TABLE MANAGER (
    MANAGER_ID INT NOT NULL GENERATED ALWAYS AS IDENTITY (START WITH 1, INCREMENT BY 1) PRIMARY KEY,
    MANAGER_PERMISSION_KEY VARCHAR(100),
    MANAGER_USER_ID VARCHAR(20)
);

CREATE TABLE ADMIN (
    ADMIN_ID INT NOT NULL GENERATED ALWAYS AS IDENTITY (START WITH 1, INCREMENT BY 1) PRIMARY KEY,
    ADMIN_PERMISSION_KEY VARCHAR(100),
    ADMIN_USER_ID VARCHAR(20)
);
